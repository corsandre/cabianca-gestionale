{# Segno del marcatore di un trattamento: pallino, riga o S nel colore del marcatore (blu, nero, rosso). Un trattamento senza segno (registrato prima del 02/11/2026) vale come pallino. #} {% macro marcatore(colore, segno, size=16) -%} {%- if colore -%} {%- set c = {'blu': '#1565c0', 'nero': '#212121', 'rosso': '#c62828'}.get(colore, '#6c757d') -%} {%- set sg = segno or 'pallino' -%} {%- set nome = {'pallino': 'pallino', 'riga': 'riga', 's': 'S'}[sg] ~ ' ' ~ (colore if colore == 'blu' or sg == 'pallino' else colore[:-1] ~ 'a') -%} {%- endif -%} {%- endmacro %}
